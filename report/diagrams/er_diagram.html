StayMart Relational Database Schema (Entity-Relationship Model)
users[Relational Node]
id (PK)BIGINT AUTO
nameVARCHAR(100)
emailVARCHAR(150) UNIQUE
password_hashVARCHAR(255)
phoneVARCHAR(20)
roleENUM(USER,ADMIN)
cityVARCHAR(100)
bioTEXT
is_activeBOOLEAN
created_atTIMESTAMP
properties[Accommodations]
id (PK)BIGINT AUTO
posted_by (FK)BIGINT -> users
titleVARCHAR(200)
descriptionTEXT
typeENUM(PG,HOSTEL..)
gender_preferenceENUM
location_id (FK)BIGINT -> loc
price_per_monthDECIMAL(10,2)
deposit_amountDECIMAL(10,2)
amenitiesJSON
imagesJSON
statusENUM(PENDING..)
rooms[Property Units]
id (PK)BIGINT AUTO
property_id (FK)BIGINT -> prop
room_numberVARCHAR(50)
typeENUM(SINGLE..)
priceDECIMAL(10,2)
capacityINT
floor_numberINT
is_availableBOOLEAN
imagesJSON
products[Marketplace]
id (PK)BIGINT AUTO
posted_by (FK)BIGINT -> users
category_id (FK)BIGINT -> cat
titleVARCHAR(200)
descriptionTEXT
priceDECIMAL(10,2)
listing_typeENUM(SELL,RENT)
conditionENUM
brandVARCHAR(100)
imagesJSON
statusENUM
bookings[Stays]
id (PK)BIGINT AUTO
property_id (FK)BIGINT
room_id (FK)BIGINT (Opt)
user_id (FK)BIGINT
check_in_dateDATE
duration_monthsINT
total_amountDECIMAL(10,2)
statusENUM(PENDING..)
orders[Transactions]
id (PK)BIGINT AUTO
product_id (FK)BIGINT
buyer_id (FK)BIGINT
order_typeENUM(BUY,RENT)
amountDECIMAL(10,2)
quantityINT
statusENUM(PENDING..)
chats[Messaging]
id (PK)BIGINT AUTO
sender_id (FK)BIGINT
receiver_id (FK)BIGINT
messageTEXT
is_readBOOLEAN
created_atTIMESTAMP
user_behaviors[ML Engine]
id (PK)BIGINT AUTO
user_id (FK)BIGINT
item_typeVARCHAR(20)
item_idBIGINT
actionENUM(VIEW,BOOK..)
weightFLOAT
created_atTIMESTAMP
